### Use cases
Bloom filters are a probabilistic data structure for set membership checks. Membership checks are `O(1)`, and they use far less memory than a hash table for large datasets.

The tradeoff is accuracy: Bloom filters can return false positives, but they never return false negatives. In other words, if it says an item is not present, that answer is guaranteed.

Instead of storing full values, a Bloom filter stores bit positions produced by multiple hash functions. Common use cases include spell checkers, banned IP lists, cache admission checks, and malicious URL checks.

For example, storing 100,000 entries with a 0.1% (`p = 0.001`) false-positive rate needs about 1,437,759 bits (~175.5 KB). That is small enough to ship to clients for local checks and reduce server work.

The sizing equations are:

`m = -(n * ln(p)) / (ln(2)^2)`

`k = (m / n) * ln(2)`

where `n` is expected entries, `p` is desired false-positive rate, `m` is number of bits, and `k` is number of hash functions.

<script is:inline>
  (() => {
    const root = document.querySelector('[data-bloom-demo]');
    if (!root) return;

    const SIZE = 24;
    const HASH_COUNT = 3;
    const inserted = ['cat', 'dog', 'bird'];
    const bits = Array(SIZE).fill(false);
    const cells = [];

    const grid = root.querySelector('[data-bit-grid]');
    const addInput = root.querySelector('[data-add-input]');
    const addButton = root.querySelector('[data-add-button]');
    const checkInput = root.querySelector('[data-check-input]');
    const addPositions = root.querySelector('[data-add-positions]');
    const checkPositions = root.querySelector('[data-check-positions]');
    const addStatus = root.querySelector('[data-add-status]');
    const checkStatus = root.querySelector('[data-check-status]');
    const insertedItems = root.querySelector('[data-inserted-items]');
    const fillRate = root.querySelector('[data-fill-rate]');
    const itemCount = root.querySelector('[data-item-count]');
    const fpRate = root.querySelector('[data-fp-rate]');

    const hashFns = [
      (value) => {
        let total = 0;
        for (let i = 0; i < value.length; i += 1) total += value.charCodeAt(i) * (i + 3);
        return total % SIZE;
      },
      (value) => {
        let total = 7;
        for (let i = 0; i < value.length; i += 1) total = (total * 31 + value.charCodeAt(i)) % 997;
        return total % SIZE;
      },
      (value) => {
        let total = 19;
        for (let i = 0; i < value.length; i += 1) total = (total + value.charCodeAt(i) * 17) % 787;
        return (total + value.length * 11) % SIZE;
      },
    ];

    function positionsFor(value) {
      return hashFns.map((fn) => fn(value));
    }

    function addValue(value) {
      positionsFor(value).forEach((position) => {
        bits[position] = true;
      });
    }

    function renderGrid(highlights = [], missing = []) {
      cells.forEach((cell, index) => {
        cell.classList.toggle('is-set', bits[index]);
        cell.classList.toggle('is-probed', highlights.includes(index));
        cell.classList.toggle('is-missing', missing.includes(index));
      });
    }

    function renderChips(target, positions, missing = []) {
      target.innerHTML = '';
      positions.forEach((position, index) => {
        const chip = document.createElement('span');
        chip.className = 'bloom-chip bloom-chip--hash';
        if (missing.includes(position)) chip.className += ' bloom-chip--missing';
        chip.textContent = `h${index + 1} -> bit ${position}`;
        target.appendChild(chip);
      });
    }

    function renderItems() {
      insertedItems.innerHTML = '';
      inserted.forEach((value) => {
        const chip = document.createElement('span');
        chip.className = 'bloom-chip';
        chip.textContent = value;
        insertedItems.appendChild(chip);
      });
    }

    function renderStats() {
      const setBits = bits.filter(Boolean).length;
      fillRate.textContent = `${setBits} / ${SIZE}`;
      itemCount.textContent = `${inserted.length}`;
      const rate = Math.pow(1 - Math.exp((-HASH_COUNT * inserted.length) / SIZE), HASH_COUNT) * 100;
      fpRate.textContent = `${rate.toFixed(1)}%`;
    }

    function previewAdd() {
      const value = addInput.value.trim().toLowerCase();
      if (!value) {
        addPositions.innerHTML = '';
        addStatus.textContent = 'Type in the field to preview which bits will flip on.';
        renderGrid();
        return;
      }

      const positions = positionsFor(value);
      renderChips(addPositions, positions);
      renderGrid(positions);
      const flipped = positions.filter((position) => !bits[position]).length;
      addStatus.textContent = flipped === 0
        ? 'All of those bits are already on, so this insert would mostly reinforce existing evidence.'
        : `${flipped} bit${flipped === 1 ? '' : 's'} would flip on for \"${value}\".`;
    }

    function previewCheck() {
      const value = checkInput.value.trim().toLowerCase();
      if (!value) {
        checkPositions.innerHTML = '';
        checkStatus.textContent = 'Probe a word to see whether the filter rejects it or returns a possible match.';
        renderGrid(addInput.value.trim() ? positionsFor(addInput.value.trim().toLowerCase()) : []);
        return;
      }

      const positions = positionsFor(value);
      const missing = positions.filter((position) => !bits[position]);
      renderChips(checkPositions, positions, missing);
      renderGrid(positions, missing);
      checkStatus.textContent = missing.length > 0
        ? `\"${value}\" is definitely not present because bit ${missing.join(', ')} is off.`
        : `\"${value}\" is possibly present because every required bit is already on.`;
    }

    for (let index = 0; index < SIZE; index += 1) {
      const cell = document.createElement('div');
      cell.className = 'bloom-demo__cell';
      cell.textContent = index;
      grid.appendChild(cell);
      cells.push(cell);
    }

    inserted.forEach(addValue);
    renderItems();
    renderStats();
    renderGrid();

    addInput.addEventListener('input', previewAdd);
    checkInput.addEventListener('input', previewCheck);
    addButton.addEventListener('click', () => {
      const value = addInput.value.trim().toLowerCase();
      if (!value) return;
      inserted.push(value);
      addValue(value);
      addInput.value = '';
      addPositions.innerHTML = '';
      addStatus.textContent = `Inserted \"${value}\" into the filter.`;
      renderItems();
      renderStats();
      previewCheck();
      renderGrid();
    });
  })();
</script>


### Python example
Here is a simple Bloom filter implementation in Python.

```python
import math
import bitarray
import mmh3


def calc_optimal_params(n_entries, false_positive_rate=0.001):
    m = (-n_entries * math.log(false_positive_rate)) / (math.log(2) ** 2)
    k = (m / n_entries) * math.log(2)
    return int(math.ceil(m)), int(math.ceil(k))


class BloomFilter:
    def __init__(self, n_entries, false_positive_rate=0.001):
        self.size, self.hash_count = calc_optimal_params(
            n_entries, false_positive_rate
        )
        self.bits = bitarray.bitarray(self.size)
        self.bits.setall(False)

    def add(self, item):
        for seed in range(self.hash_count):
            pos = mmh3.hash(item, seed) % self.size
            self.bits[pos] = True

    def contains(self, item):
        for seed in range(self.hash_count):
            pos = mmh3.hash(item, seed) % self.size
            if not self.bits[pos]:
                return False
        return True

def load_words(path='/usr/share/dict/words'):
    with open(path, 'r') as f:
        return [word.strip() for word in f]


if __name__ == '__main__':
    words = load_words()
    bloom = BloomFilter(n_entries=len(words), false_positive_rate=0.001)

    for word in words:
        bloom.add(word)

    tests = ['badwordforsure', 'cat', 'hello', 'jsalj']
    for t in tests:
        state = 'possibly present' if bloom.contains(t) else 'definitely not present'
        print(t, '->', state)
```
